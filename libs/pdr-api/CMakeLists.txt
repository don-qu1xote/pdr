# ТИПЫ КОНТРАКТА — ИЗ СПЕЦИФИКАЦИИ, А НЕ ИЗ ГОЛОВЫ.
#
# Здесь нет ни одного написанного руками типа, и это весь смысл каталога.
# Структуры, разборщики и сериализаторы для тел запросов и ответов порождает
# штатный chaotic из docs/api/openapi.yaml на этапе сборки (ADR-0013).
#
# ПОРОЖДЁННОЕ НЕ КОММИТИТСЯ. Оно живёт в каталоге сборки и появляется заново у
# каждого, кто собирает: закоммиченный файл — это копия, а копия расходится с
# оригиналом в тот день, когда правку внесли в один из них. Ровно поэтому
# расхождение схемы и кода здесь невыразимо: кода, который мог бы разойтись,
# нет.
#
# Обратная сторона правила — фронт: у него своего chaotic нет, и порождённое
# ему приходится коммитить. Асимметрия названа и обоснована в docs/api/README.md.
#
# Без userver порождать нечем: chaotic — его часть. Тогда каталог не собирается,
# как и весь слой адаптеров.
if(NOT userver_FOUND)
    return()
endif()

# ИМЕНА ТИПОВ ЗАДАЁТ СХЕМА, А НЕ ЭТОТ ФАЙЛ: `/components/schemas/SignInRequest`
# становится `pdr::api::SignInRequest`. Переименовать тип, не переименовав
# схему, нельзя — правило и есть отображение.
#
# GENERATE_SERIALIZERS нужен потому, что ручка не только читает тела, но и
# отдаёт: без него порождается половина круга.
userver_target_generate_chaotic(pdr_api
    OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated"
    OUTPUT_PREFIX pdr/api
    RELATIVE_TO "${CMAKE_SOURCE_DIR}/docs/api"
    LAYOUT "/components/schemas/([^/]*)/=pdr::api::{0}"
    GENERATE_SERIALIZERS
    SCHEMAS "${CMAKE_SOURCE_DIR}/docs/api/openapi.yaml"
)
